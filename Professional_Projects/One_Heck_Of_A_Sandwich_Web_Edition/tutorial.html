Java 
public class Order {
    private ArrayList<Product> products;

    public Order() {
        this.products = new ArrayList<>();
    }

    public void addProduct(Product product) {
        products.add(product);
        System.out.println("Added: "
            + product.getName());
    }

    public double getTotal() {
        double total = 0;
        for (Product p : products) {
            total += p.getPrice();
        }
        return total;
    }
}
JavaScript 
class Order {
    constructor() {
        this.products = [];
    }

    addProduct(product) {
        this.products.push(product);
        this.updateOrderDisplay();
        this.showAddedAnimation(product);
    }

    getTotal() {
        return this.products.reduce(
            (sum, p) => sum + p.price, 0
        );
    }

    updateOrderDisplay() {
        document.getElementById('order-items')
            .innerHTML = this.renderItems();
    }
}

public class Sandwich extends Product {
    private int size;
    private String breadType;
    private boolean toasted;
    private ArrayList<Topping> toppings;

    public Sandwich(int size, String bread) {
        this.size = size;
        this.breadType = bread;
        this.toppings = new ArrayList<>();
        this.toasted = false;
    }

    public void addTopping(Topping t) {
        toppings.add(t);
        System.out.println("Added "
            + t.getName());
    }
}
JavaScript 
class Sandwich extends Product {
    constructor(size, bread) {
        super();
        this.size = size;
        this.breadType = bread;
        this.toppings = [];
        this.toasted = false;
    }

    addTopping(topping) {
        this.toppings.push(topping);

        // Update UI with animation
        const elem = document.querySelector(
            `[data-topping="${topping.name}"]`
        );
        elem.classList.add('selected');
    }
}
Java 
public void checkout() {
    System.out.println("=== RECEIPT ===");

    for (Product p : products) {
        System.out.println(
            p.getName() + " - $"
            + String.format("%.2f", p.getPrice())
        );
    }

    System.out.println("-------------");
    System.out.println("TOTAL: $"
        + String.format("%.2f", getTotal()));

    saveToFile();
}
JavaScript 
function checkout() {
    const receipt = document.createElement('div');
    receipt.className = 'receipt-kiosk';

    receipt.innerHTML = `
        <h3>Your Order</h3>
        ${currentOrder.products.map(p => `
            <div class="receipt-item">
                <span>${p.name}</span>
                <span>$${p.price.toFixed(2)}</span>
            </div>
        `).join('')}
        <div class="receipt-total">
            TOTAL: $${currentOrder.getTotal()}
        </div>
    `;

    showScreen('checkout');
}

Key Differences

Input/Output CLI uses Scanner & System.out → Web uses DOM manipulation
User Feedback Text confirmations → Visual animations & transitions
State Management Simple variables → Real-time UI updates
